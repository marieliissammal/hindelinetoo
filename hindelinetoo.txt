import math #lisasin import, siis programm saab teha matemaatikat.
print("Ruutvõrrandi lahendamine") # " asemel lõpus oli '. Kuid see on võimalik, see ainult sobib siis kui mõlemad on sarnased. 
a=float(input("Sisesta a => "))
b=float(input("Sisesta b => ")) # Polnud ) lõpus 
c=float(input("Sisesta c => ")) # polnud float 
D=b*(b-4*a*c) # polnud sulgudes
if D>0: # polnud : lõpus
    x1=(-b+D**0.5)/(2*a) # 2*a polnud sulgudes
    x2=(-b-D**0.5)/(2*a)
    print("Kaks lahendit:", x1, x2)
else if D==0: # 2 '=' näitab, et see võrdub millegiga
    x=-b/(2*a)
    print("Üks lahend:", x)
else:
print("Reaalarvulised lahendid puuduvad")
print()
